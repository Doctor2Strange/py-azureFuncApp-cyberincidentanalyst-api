# ---------------------------------------------------------
# Helper: create standard JSON HTTP responses
# ---------------------------------------------------------

def json_response(
    body: dict,
    status_code: int = 200,
) -> func.HttpResponse:

    return func.HttpResponse(
        json.dumps(
            body,
            ensure_ascii=False,
        ),
        status_code=status_code,
        mimetype="application/json",
    )


# ---------------------------------------------------------
# Helper: remove accidental Markdown code fences
#
# The system prompt tells the model not to use them,
# but this protects the API if it does anyway.
# ---------------------------------------------------------

def strip_code_fences(
    text: str,
) -> str:

    text = text.strip()

    if text.startswith("```json"):
        text = text[7:]

    elif text.startswith("```"):
        text = text[3:]

    if text.endswith("```"):
        text = text[:-3]

    return text.strip()


# ---------------------------------------------------------
# Helper: translate numeric confidence into a band
#
# IMPORTANT:
# The Function calculates this itself.
# The LLM does not decide the confidence band.
# ---------------------------------------------------------

def get_confidence_band(
    score: float,
) -> str:

    if score < 0.40:
        return "low"

    if score < 0.70:
        return "medium"

    if score < 0.90:
        return "high"

    return "very_high"


# ---------------------------------------------------------
# Helper: validate model confidence
# ---------------------------------------------------------

def validate_confidence(
    assessment: dict,
) -> tuple[bool, str | None]:

    confidence = assessment.get(
        "confidence"
    )

    if not isinstance(
        confidence,
        (int, float),
    ):
        return (
            False,
            "Model returned a non-numeric confidence value",
        )

    if confidence < 0 or confidence > 1:
        return (
            False,
            "Model confidence must be between 0 and 1",
        )

    return (
        True,
        None,
    )

# ---------------------------------------------------------
# POST /api/v1/security/analyze-alert
#
# Main SIEM-facing endpoint.
# ---------------------------------------------------------

@app.route(
    route="v1/security/analyze-alert",
    methods=["POST"],
)
def analyze_alert(
    req: func.HttpRequest,
) -> func.HttpResponse:

    # -----------------------------------------------------
    # Correlation ID
    #
    # Preserve caller-supplied ID when available.
    # Otherwise create one.
    # -----------------------------------------------------

    correlation_id = (
        req.headers.get(
            "X-Correlation-ID"
        )
        or str(uuid.uuid4())
    )


    # -----------------------------------------------------
    # Parse incoming JSON
    # -----------------------------------------------------

    try:

        request_data = (
            req.get_json()
        )

    except ValueError:

        return json_response(
            {
                "error":
                    "Request body must be valid JSON",

                "correlation_id":
                    correlation_id,
            },
            400,
        )


    # -----------------------------------------------------
    # Basic API request validation
    # -----------------------------------------------------

    if (
        request_data.get(
            "schema_version",
            "1.0",
        )
        != "1.0"
    ):

        return json_response(
            {
                "error":
                    "Unsupported schema_version",

                "correlation_id":
                    correlation_id,
            },
            400,
        )


    incident_id = (
        request_data.get(
            "incident_id"
        )
    )


    if not incident_id:

        return json_response(
            {
                "error":
                    "incident_id is required",

                "correlation_id":
                    correlation_id,
            },
            400,
        )


    alert = request_data.get(
        "alert"
    )


    if not isinstance(
        alert,
        dict,
    ):

        return json_response(
            {
                "error":
                    "alert must be a JSON object",

                "correlation_id":
                    correlation_id,
            },
            400,
        )


    if not alert.get(
        "title"
    ):

        return json_response(
            {
                "error":
                    "alert.title is required",

                "correlation_id":
                    correlation_id,
            },
            400,
        )


    task = request_data.get(
        "task",
        "triage",
    )


    # -----------------------------------------------------
    # Construct the document that will be sent to the LLM
    # -----------------------------------------------------

    model_input = {
        "incident_id":
            incident_id,

        "task":
            task,

        "alert":
            alert,
    }


    # -----------------------------------------------------
    # Authenticate to Azure Foundry
    #
    # No API key is required.
    #
    # The Function's system-assigned Managed Identity
    # obtains a Microsoft Entra access token.
    # -----------------------------------------------------

    try:

        token = (
            credential.get_token(
                "https://cognitiveservices.azure.com/.default"
            )
        )

    except Exception:

        # Do NOT expose detailed authentication errors
        # to the API caller.

        return json_response(
            {
                "error":
                    "Unable to authenticate to model backend",

                "correlation_id":
                    correlation_id,
            },
            502,
        )


    # -----------------------------------------------------
    # Create Azure OpenAI / Foundry SDK client
    #
    # MODEL_BASE_URL example:
    #
    # https://resource.openai.azure.com/openai/v1
    #
    # AzureOpenAI expects:
    #
    # https://resource.openai.azure.com
    # -----------------------------------------------------

    try:

        azure_endpoint = (
            MODEL_BASE_URL
            .split(
                "/openai/v1"
            )[0]
        )


        client = AzureOpenAI(
            azure_endpoint=
                azure_endpoint,

            api_version=
                "2024-10-21",

            azure_ad_token=
                token.token,

            timeout=
                MODEL_TIMEOUT_SECONDS,
        )


        # -------------------------------------------------
        # Send request to model
        #
        # MODEL_NAME must be the Azure deployment name.
        # -------------------------------------------------

        response = (
            client
            .chat
            .completions
            .create(
                model=
                    MODEL_NAME,

                messages=[
                    {
                        "role":
                            "system",

                        "content":
                            SYSTEM_PROMPT,
                    },
                    {
                        "role":
                            "user",

                        "content":
                            json.dumps(
                                model_input,
                                ensure_ascii=False,
                            ),
                    },
                ],

                # Newer models may require
                # max_completion_tokens instead
                # of max_tokens.

                max_completion_tokens=
                    700,
            )
        )


    except Exception:

        # Detailed upstream errors should be recorded
        # internally in production, for example through
        # Application Insights.
        #
        # Do not expose implementation details to callers.

        return json_response(
            {
                "error":
                    "Model backend request failed",

                "correlation_id":
                    correlation_id,
            },
            502,
        )


    # -----------------------------------------------------
    # Extract model output
    # -----------------------------------------------------

    try:

        raw_content = (
            response
            .choices[0]
            .message
            .content
        )


        if not raw_content:

            return json_response(
                {
                    "error":
                        "Model returned an empty response",

                    "correlation_id":
                        correlation_id,
                },
                502,
            )


        raw_content = (
            strip_code_fences(
                raw_content
            )
        )


        assessment = (
            json.loads(
                raw_content
            )
        )


    except (
        json.JSONDecodeError,
        IndexError,
        AttributeError,
        TypeError,
    ):

        return json_response(
            {
                "error":
                    (
                        "Model did not return "
                        "valid structured JSON"
                    ),

                "correlation_id":
                    correlation_id,
            },
            502,
        )


    # -----------------------------------------------------
    # Validate confidence returned by the model
    # -----------------------------------------------------

    confidence_valid, confidence_error = (
        validate_confidence(
            assessment
        )
    )


    if not confidence_valid:

        return json_response(
            {
                "error":
                    confidence_error,

                "correlation_id":
                    correlation_id,
            },
            502,
        )


    confidence = float(
        assessment[
            "confidence"
        ]
    )


    # -----------------------------------------------------
    # Add confidence interpretation ourselves
    #
    # The LLM does NOT determine the confidence band.
    # -----------------------------------------------------

    assessment[
        "confidence_band"
    ] = get_confidence_band(
        confidence
    )


    assessment[
        "confidence_scale"
    ] = {

        "low":
            "0.00-0.39",

        "medium":
            "0.40-0.69",

        "high":
            "0.70-0.89",

        "very_high":
            "0.90-1.00",

        "interpretation":
            (
                "Confidence represents the strength "
                "and consistency of the supplied evidence."
            ),

        "important_note":
            (
                "The confidence score is generated by "
                "the language model and is not a "
                "mathematically calibrated probability."
            ),
    }


    # -----------------------------------------------------
    # Return stable SIEM-facing response
    # -----------------------------------------------------

    return json_response(
        {
            "schema_version":
                "1.0",

            "incident_id":
                incident_id,

            "correlation_id":
                correlation_id,

            "assessment":
                assessment,

            "model": {

                "provider":
                    MODEL_PROVIDER,

                "name":
                    MODEL_NAME,
            },

            "warning":
                (
                    "AI assessment is advisory and "
                    "requires human analyst verification."
                ),
        }
    )
